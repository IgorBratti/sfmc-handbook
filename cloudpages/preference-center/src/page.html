%%[ /* Preference center with "unsubscribe from all" Link from the email footer (see README). Parameters travel encrypted and the subscriber key is signed: RedirectTo(CloudPagesURL(3456, "sk", _subscriberkey, "sig", @sig, "jid", jobid, "lid", listid, "bid", _JobSubscriberBatchID)) - Without a valid signature the page shows "This link is not valid" and does nothing (see ../../security.md). - "Save" updates topic preferences in Preferences_DE. - "Unsubscribe from all" clears every topic and logs a real unsubscribe with LogUnsubEvent, tied to the email the contact came from (JobID, ListID, BatchID) so it appears in tracking. - The result of LogUnsubEvent is checked; failures are logged to Error_Log_DE and the visitor sees a neutral message. The form posts back to the same URL, so the encrypted parameters stay with it. */ VAR @sk, @sig, @secret, @jid, @lid, @bid, @action, @news, @offers, @events VAR @message, @messageType, @lue, @prop, @statusCode, @statusMsg, @requestId SET @sk = RequestParameter("sk") SET @sig = RequestParameter("sig") SET @jid = RequestParameter("jid") SET @lid = RequestParameter("lid") SET @bid = RequestParameter("bid") SET @action = RequestParameter("action") SET @secret = Lookup("Config_DE", "Value", "Name", "LINK_SIGNING_SECRET") IF Empty(@sk) OR Empty(@secret) OR @sig != SHA256(Concat(@sk, "|", @secret), "UTF-8") THEN SET @sk = "" ENDIF IF NOT Empty(@sk) THEN IF @action == "save" THEN SET @news = IIF(RequestParameter("newsletter") == "on", "true", "false") SET @offers = IIF(RequestParameter("offers") == "on", "true", "false") SET @events = IIF(RequestParameter("events") == "on", "true", "false") UpsertData("Preferences_DE", 1, "SubscriberKey", @sk, "Newsletter", @news, "Offers", @offers, "Events", @events, "UpdatedAt", Now()) SET @message = "Your preferences were saved." SET @messageType = "ok" ELSEIF @action == "unsubscribe_all" THEN UpsertData("Preferences_DE", 1, "SubscriberKey", @sk, "Newsletter", "false", "Offers", "false", "Events", "false", "UpdatedAt", Now()) SET @lue = CreateObject("ExecuteRequest") SetObjectProperty(@lue, "Name", "LogUnsubEvent") SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "SubscriberKey") SetObjectProperty(@prop, "Value", @sk) AddObjectArrayItem(@lue, "Parameters", @prop) IF NOT Empty(@jid) THEN SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "JobID") SetObjectProperty(@prop, "Value", @jid) AddObjectArrayItem(@lue, "Parameters", @prop) ENDIF IF NOT Empty(@lid) THEN SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "ListID") SetObjectProperty(@prop, "Value", @lid) AddObjectArrayItem(@lue, "Parameters", @prop) ENDIF IF NOT Empty(@bid) THEN SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "BatchID") SetObjectProperty(@prop, "Value", @bid) AddObjectArrayItem(@lue, "Parameters", @prop) ENDIF SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "Reason") SetObjectProperty(@prop, "Value", "Unsubscribed from all via preference center") AddObjectArrayItem(@lue, "Parameters", @prop) SET @statusCode = InvokeExecute(@lue, @statusMsg, @requestId) IF @statusCode == "OK" THEN SET @message = "You have been unsubscribed from all emails." SET @messageType = "ok" ELSE InsertData("Error_Log_DE", "Source", "PreferenceCenter_LogUnsubEvent", "Message", Concat(@statusCode, " | ", @statusMsg, " | ", @requestId), "CreatedAt", Now()) SET @message = "We couldn't complete your request. Please try again in a few minutes." SET @messageType = "error" ENDIF ENDIF SET @news = Lookup("Preferences_DE", "Newsletter", "SubscriberKey", @sk) SET @offers = Lookup("Preferences_DE", "Offers", "SubscriberKey", @sk) SET @events = Lookup("Preferences_DE", "Events", "SubscriberKey", @sk) ENDIF ]%% Email preferences
