%%[ /* Preference center with "unsubscribe from all" Link from the email footer (parameters travel encrypted, so nobody can edit someone else's preferences): RedirectTo(CloudPagesURL(3456, "sk", _subscriberkey, "jid", jobid, "lid", listid, "bid", _JobSubscriberBatchID)) - "Save" updates topic preferences in Preferences_DE. - "Unsubscribe from all" clears every topic and logs a real unsubscribe with LogUnsubEvent, tied to the email the contact came from (JobID, ListID, BatchID) for correct tracking. The form posts back to the same URL, so the encrypted parameters stay with it. */ VAR @sk, @jid, @lid, @bid, @action, @news, @offers, @events, @message, @lue, @prop, @statusCode, @statusMsg, @requestId SET @sk = RequestParameter("sk") SET @jid = RequestParameter("jid") SET @lid = RequestParameter("lid") SET @bid = RequestParameter("bid") SET @action = RequestParameter("action") IF NOT Empty(@sk) THEN IF @action == "save" THEN SET @news = IIF(RequestParameter("newsletter") == "on", "true", "false") SET @offers = IIF(RequestParameter("offers") == "on", "true", "false") SET @events = IIF(RequestParameter("events") == "on", "true", "false") UpsertData("Preferences_DE", 1, "SubscriberKey", @sk, "Newsletter", @news, "Offers", @offers, "Events", @events, "UpdatedAt", Now()) SET @message = "Your preferences were saved." ELSEIF @action == "unsubscribe_all" THEN UpsertData("Preferences_DE", 1, "SubscriberKey", @sk, "Newsletter", "false", "Offers", "false", "Events", "false", "UpdatedAt", Now()) SET @lue = CreateObject("ExecuteRequest") SetObjectProperty(@lue, "Name", "LogUnsubEvent") SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "SubscriberKey") SetObjectProperty(@prop, "Value", @sk) AddObjectArrayItem(@lue, "Parameters", @prop) IF NOT Empty(@jid) THEN SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "JobID") SetObjectProperty(@prop, "Value", @jid) AddObjectArrayItem(@lue, "Parameters", @prop) ENDIF IF NOT Empty(@lid) THEN SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "ListID") SetObjectProperty(@prop, "Value", @lid) AddObjectArrayItem(@lue, "Parameters", @prop) ENDIF IF NOT Empty(@bid) THEN SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "BatchID") SetObjectProperty(@prop, "Value", @bid) AddObjectArrayItem(@lue, "Parameters", @prop) ENDIF SET @prop = CreateObject("APIProperty") SetObjectProperty(@prop, "Name", "Reason") SetObjectProperty(@prop, "Value", "Unsubscribed from all via preference center") AddObjectArrayItem(@lue, "Parameters", @prop) SET @statusCode = InvokeExecute(@lue, @statusMsg, @requestId) SET @message = "You have been unsubscribed from all emails." ENDIF SET @news = Lookup("Preferences_DE", "Newsletter", "SubscriberKey", @sk) SET @offers = Lookup("Preferences_DE", "Offers", "SubscriberKey", @sk) SET @events = Lookup("Preferences_DE", "Events", "SubscriberKey", @sk) ENDIF ]%% Email preferences
