%%[ /* Birthday page with a unique coupon per contact. Link from the birthday email (see README): RedirectTo(CloudPagesURL(2345, "sk", _subscriberkey, "sig", @sig)) - The subscriber key is accepted only with a valid signature, so nobody can claim coupons for other keys by editing the URL (see ../../security.md). - ClaimRowValue gives this contact one unused coupon and marks it as claimed. Opening the page again returns the same coupon, so refreshes and link scanners are safe. */ VAR @sk, @sig, @secret, @firstName, @coupon, @expiresAt, @expires, @hasCoupon SET @sk = RequestParameter("sk") SET @sig = RequestParameter("sig") SET @hasCoupon = "false" SET @secret = Lookup("Config_DE", "Value", "Name", "LINK_SIGNING_SECRET") IF Empty(@sk) OR Empty(@secret) OR @sig != SHA256(Concat(@sk, "|", @secret), "UTF-8") THEN SET @sk = "" ENDIF IF NOT Empty(@sk) THEN SET @firstName = RegExMatch(Lookup("Customers_DE", "FirstName", "SubscriberKey", @sk), "^[\p{L} '-]{1,50}$", 0) SET @coupon = ClaimRowValue("Birthday_Coupons_DE", "CouponCode", "IsClaimed", "", "SubscriberKey", @sk) IF NOT Empty(@coupon) THEN SET @expiresAt = Lookup("Birthday_Coupons_DE", "ExpiresAt", "CouponCode", @coupon) SET @expires = FormatDate(@expiresAt, "MMMM d, yyyy", "", "en-US") SET @hasCoupon = "true" ENDIF ENDIF IF Empty(@firstName) THEN SET @firstName = "friend" ELSE SET @firstName = ProperCase(@firstName) ENDIF ]%% Happy birthday

BrandName

Happy birthday, %%=v(@firstName)=%%!

%%[ IF @hasCoupon == "true" THEN ]%%

To celebrate, here is a gift just for you: 20% off your next purchase.

%%=v(@coupon)=%%

Valid until %%=v(@expires)=%%. One use per customer.

Use my gift %%[ ELSE ]%%

We wish you a wonderful day. Keep an eye on your inbox: something special is coming.

Visit our website %%[ ENDIF ]%%
