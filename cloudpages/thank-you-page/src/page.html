%%[ /* Thank-you page Two ways to reach it, both through CloudPagesURL (encrypted query string): after a form post ..... Redirect(CloudPagesURL(1234, "fn", @firstName)) from an email ......... RedirectTo(CloudPagesURL(1234, "sk", _subscriberkey, "sig", @sig)) - "fn" is printed on the page, so it is allow-listed (letters, spaces, apostrophes, hyphens). - "sk" is an identity, so it is accepted only with a valid signature (see ../../security.md). Replace 1234 with this page's ID. */ VAR @firstName, @sk, @sig, @secret, @greeting SET @firstName = RegExMatch(RequestParameter("fn"), "^[\p{L} '-]{1,50}$", 0) SET @sk = RequestParameter("sk") SET @sig = RequestParameter("sig") IF Empty(@firstName) AND NOT Empty(@sk) THEN SET @secret = Lookup("Config_DE", "Value", "Name", "LINK_SIGNING_SECRET") IF NOT Empty(@secret) AND @sig == SHA256(Concat(@sk, "|", @secret), "UTF-8") THEN SET @firstName = RegExMatch(Lookup("Customers_DE", "FirstName", "SubscriberKey", @sk), "^[\p{L} '-]{1,50}$", 0) ENDIF ENDIF IF Empty(@firstName) THEN SET @greeting = "Thank you!" ELSE SET @greeting = Concat("Thank you, ", ProperCase(@firstName), "!") ENDIF ]%% Thank you
