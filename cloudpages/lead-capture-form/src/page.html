%%[ /* Lead capture form (click-to-lead) - Writes only on a real submission with valid input (AMPscript runs on every render). - Every value that is stored or printed is validated or allow-listed (see ../../security.md). - Hidden honeypot field filters simple bots without a CAPTCHA. - First-touch attribution: Source and CreatedAt are written once; LastSource and UpdatedAt on every submission. - Consent and its timestamp are stored as LGPD/GDPR evidence. - Dates use server time (US Central, UTC-6, no daylight saving). Convert when reporting. On success it redirects to the thank-you page. Replace 1234 with that page's ID. */ VAR @submitted, @firstName, @lastName, @email, @phone, @interest, @consent, @honeypot VAR @source, @error, @now, @existing SET @submitted = RequestParameter("submitted") /* Allow-list the traffic source: it is printed into a hidden field and stored. */ SET @source = RegExMatch(RequestParameter("utm_source"), "^[A-Za-z0-9_.-]{1,50}$", 0) IF Empty(@source) THEN SET @source = "direct" ENDIF IF @submitted == "1" THEN SET @firstName = Trim(RequestParameter("firstName")) SET @lastName = Trim(RequestParameter("lastName")) SET @email = Lowercase(Trim(RequestParameter("email"))) SET @phone = Replace(Replace(Replace(Replace(Trim(RequestParameter("phone")), " ", ""), "-", ""), "(", ""), ")", "") SET @interest = RequestParameter("interest") SET @consent = RequestParameter("consent") SET @honeypot = RequestParameter("website") /* Closed list: only the options the form offers. */ IF @interest != "Product A" AND @interest != "Product B" THEN SET @interest = "Other" ENDIF IF NOT Empty(@honeypot) THEN /* A bot filled the hidden field: show the normal success page, store nothing. */ Redirect(CloudPagesURL(1234)) ELSEIF Empty(RegExMatch(@firstName, "^[\p{L} '-]{1,100}$", 0)) OR Empty(RegExMatch(@lastName, "^[\p{L} '-]{1,100}$", 0)) OR NOT IsEmailAddress(@email) THEN SET @error = "Please fill in your name and a valid email address." ELSEIF NOT Empty(@phone) AND Empty(RegExMatch(@phone, "^\+?[0-9]{8,15}$", 0)) THEN SET @error = "Please enter a valid phone number, or leave it empty." ELSEIF @consent != "on" THEN SET @error = "Please accept the privacy terms to continue." ELSE SET @now = Now() SET @existing = Lookup("Leads_DE", "EmailAddress", "EmailAddress", @email) IF Empty(@existing) THEN InsertData("Leads_DE", "EmailAddress", @email, "FirstName", @firstName, "LastName", @lastName, "Phone", @phone, "Interest", @interest, "Consent", "true", "ConsentAt", @now, "Source", @source, "LastSource", @source, "CreatedAt", @now, "UpdatedAt", @now) ELSE UpdateData("Leads_DE", 1, "EmailAddress", @email, "FirstName", @firstName, "LastName", @lastName, "Phone", @phone, "Interest", @interest, "Consent", "true", "ConsentAt", @now, "LastSource", @source, "UpdatedAt", @now) ENDIF /* Optional, with Marketing Cloud Connect: create the lead in Sales Cloud too. SET @leadId = CreateSalesforceObject("Lead", 5, "FirstName", @firstName, "LastName", @lastName, "Email", @email, "Company", "Website lead", "LeadSource", "CloudPage") */ Redirect(CloudPagesURL(1234, "fn", @firstName)) ENDIF ENDIF ]%% Talk to us
