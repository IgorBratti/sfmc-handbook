On success it redirects to the thank-you page. Replace 1234 with that page's ID. */ VAR @submitted, @firstName, @lastName, @email, @phone, @interest, @consent, @honeypot, @source, @error SET @submitted = RequestParameter("submitted") SET @source = RequestParameter("utm_source") IF Empty(@source) THEN SET @source = "direct" ENDIF IF @submitted == "1" THEN SET @firstName = Trim(RequestParameter("firstName")) SET @lastName = Trim(RequestParameter("lastName")) SET @email = Lowercase(Trim(RequestParameter("email"))) SET @phone = Replace(Replace(Replace(Replace(Trim(RequestParameter("phone")), " ", ""), "-", ""), "(", ""), ")", "") SET @interest = RequestParameter("interest") SET @consent = RequestParameter("consent") SET @honeypot = RequestParameter("website") IF NOT Empty(@honeypot) THEN /* A bot filled the hidden field: show success, store nothing. */ Redirect(CloudPagesURL(1234)) ELSEIF Empty(@firstName) OR Empty(@lastName) OR NOT IsEmailAddress(@email) THEN SET @error = "Please fill in your name and a valid email address." ELSEIF @consent != "on" THEN SET @error = "Please accept the privacy terms to continue." SET @leadId = CreateSalesforceObject("Lead", 5, "FirstName", @firstName, "LastName", @lastName, "Email", @email, "Company", "Website lead", "LeadSource", "CloudPage") */ Redirect(CloudPagesURL(1234, "fn", @firstName)) ENDIF ENDIF ]%% Talk to us
